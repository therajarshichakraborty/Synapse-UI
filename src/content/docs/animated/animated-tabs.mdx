## Usage

```tsx
import { AnimatedTabs } from "@/components/synapse-ui/animated-tabs";
```

```tsx
<AnimatedTabs
  tabs={[
    { value: "a", label: "Product", content: <Panel /> },
    { value: "b", label: "Services", content: <Panel /> },
  ]}
/>
```
