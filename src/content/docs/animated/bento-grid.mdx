## Usage

```tsx
import { BentoCard, BentoGrid } from "@/components/synapse-ui/bento-grid";
```

```tsx
<BentoGrid>
  <BentoCard
    className="md:col-span-2"
    title="AI"
    description="..."
    icon={<SparklesIcon />}
    header={<Visual />}
  />
  <BentoCard title="Secure" description="..." />
</BentoGrid>
```
