## Usage

```tsx
import { Terminal } from "@/components/animated/terminal";
```

```tsx
<Terminal
  lines={[
    { type: "command", text: "npx synapse-ui add button" },
    { type: "output", text: "✔ Done", tone: "success" },
  ]}
/>
```
