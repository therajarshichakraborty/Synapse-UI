## Usage

```tsx
import { DropdownMenu } from "@/components/native/dropdown-menu";
```

```tsx
<DropdownMenu items={[{ label: "Edit", icon: "edit", onPress: edit }]}>
  {(open) => <IconButton icon="more-horizontal" accessibilityLabel="More" onPress={open} />}
</DropdownMenu>
```
