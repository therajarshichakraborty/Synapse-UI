## Installation

<ComponentInstall name="native-chip" />

## Usage

```tsx
import { Chip, ChipGroup } from "@/components/native/chip";
```

```tsx
<ChipGroup
  multiple
  options={[{ value: "a", label: "Design" }]}
  value={tags}
  onValueChange={setTags}
/>
```
