## Installation

<ComponentInstall name="native-avatar" />

## Usage

```tsx
import { Avatar, AvatarGroup } from "@/components/native/avatar";
```

```tsx
<Avatar src={user.photo} name={user.name} status="online" />
<AvatarGroup people={members} max={4} />
```
