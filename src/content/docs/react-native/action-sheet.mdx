## Usage

```tsx
import { ActionSheet } from "@/components/native/action-sheet";
```

```tsx
<ActionSheet
  open={open}
  onOpenChange={setOpen}
  options={[{ label: "Delete", destructive: true, onPress }]}
/>
```
