## Usage

```tsx
import { RadioGroup } from "@/components/native/radio-group";
```

```tsx
<RadioGroup
  variant="card"
  value={plan}
  onValueChange={setPlan}
  options={[{ value: "a", label: "Option A" }]}
/>
```
