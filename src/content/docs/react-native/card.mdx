## Usage

```tsx
import { Card, CardContent, CardFooter, CardHeader } from "@/components/native/card";
```

```tsx
<Card variant="elevated">
  <CardHeader title="Title" description="Description" />
  <CardContent>…</CardContent>
  <CardFooter>…</CardFooter>
</Card>
```
