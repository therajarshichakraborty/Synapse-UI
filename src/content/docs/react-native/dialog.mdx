## Usage

```tsx
import { Dialog } from "@/components/native/dialog";
```

```tsx
<Dialog
  open={open}
  onOpenChange={setOpen}
  title="Rename"
  footer={<Button onPress={save}>Save</Button>}
>
  <Input />
</Dialog>
```
