## Usage

```tsx
import { ListGroup, ListItem } from "@/components/native/list";
```

```tsx
<ListGroup header="General">
  <ListItem title="Wi-Fi" icon="wifi" iconColor="#3b82f6" value="Home" onPress={open} />
  <ListItem title="Dark mode" icon="moon" trailing={<Switch />} />
</ListGroup>
```
