title: Dark Mode
description: Dark is the default experience. Light is just as polished. Both switch without a flash.
---

Synapse UI switches themes with a `dark` class on `<html>`. Tokens are redefined under `.dark`, so every component follows automatically.

<Steps>

### Install next-themes

<PmCommand install="next-themes" />

### Create a theme provider

```tsx title="components/theme-provider.tsx"
"use client";

import { ThemeProvider as NextThemesProvider } from "next-themes";

export function ThemeProvider(props: React.ComponentProps<typeof NextThemesProvider>) {
  return <NextThemesProvider {...props} />;
}
```

### Wrap your root layout

```tsx title="app/layout.tsx" {7-9}
import { ThemeProvider } from "@/components/theme-provider";

export default function RootLayout({ children }: { children: React.ReactNode }) {
  return (
    <html lang="en" suppressHydrationWarning>
      <body>
        <ThemeProvider attribute="class" defaultTheme="dark" enableSystem disableTransitionOnChange>
          {children}
        </ThemeProvider>
      </body>
    </html>
  );
}
```

### Add a toggle

```tsx title="components/theme-toggle.tsx"
"use client";

import { MoonIcon, SunIcon } from "lucide-react";
import { useTheme } from "next-themes";
import { Button } from "@/components/ui/button";

export function ThemeToggle() {
  const { resolvedTheme, setTheme } = useTheme();
  return (
    <Button
      variant="ghost"
      size="icon"
      aria-label="Toggle theme"
      onClick={() => setTheme(resolvedTheme === "dark" ? "light" : "dark")}
    >
      <SunIcon className="scale-100 dark:scale-0" />
      <MoonIcon className="absolute scale-0 dark:scale-100" />
    </Button>
  );
}
```

</Steps>

## Scoped themes

You can render a subtree in a fixed theme by adding `light` or `dark` to any element. Tokens and `dark:` variants inside it follow that element, not the page. The preview canvases on this site use this.

```tsx
<section className="light bg-background text-foreground">
  {/* Always light, even when the page is dark */}
</section>
```

<Callout title="How the dark variant works">
  The `dark:` variant is defined as `&:where(.dark, .dark *):not(:where(.light, .light *))`, so a
  nested `.light` scope switches `dark:` utilities off.
</Callout>
