## Usage

```tsx
import { Calendar } from "@/components/synapse-ui/calendar";
```

```tsx
const [date, setDate] = React.useState<Date | undefined>(new Date());

<Calendar mode="single" selected={date} onSelect={setDate} className="rounded-xl border" />;
```
