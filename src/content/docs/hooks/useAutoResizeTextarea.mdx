# useAutoResizeTextarea

A client-side hook that dynamically adjusts textarea height as the user types, with configurable minimum and maximum bounds.

## Import

```tsx
import { useAutoResizeTextarea } from "@/hooks/use-auto-resize-textarea";
```

## Manual Installation

If you prefer to install manually, create a file named `src/hooks/use-auto-resize-textarea.ts` and paste the following code:

```tsx
"use client";

import * as React from "react";

interface UseAutoResizeTextareaProps {
  minHeight?: number;
  maxHeight?: number;
}

export function useAutoResizeTextarea({
  minHeight = 52,
  maxHeight = 200,
}: UseAutoResizeTextareaProps = {}) {
  const textareaRef = React.useRef<HTMLTextAreaElement>(null);

  const adjustHeight = React.useCallback(
    (reset?: boolean) => {
      const textarea = textareaRef.current;
      if (!textarea) return;

      if (reset) {
        textarea.style.height = `${minHeight}px`;
        return;
      }

      textarea.style.height = "auto";
      const newHeight = Math.min(Math.max(textarea.scrollHeight, minHeight), maxHeight);
      textarea.style.height = `${newHeight}px`;
    },
    [minHeight, maxHeight],
  );

  return { textareaRef, adjustHeight };
}
```

## API

| Option | Type | Default | Description |
| ------ | ---- | ------- | ----------- |
| `minHeight` | `number` | `52` | Minimum height in pixels |
| `maxHeight` | `number` | `200` | Maximum height in pixels |

## Returns

| Property | Type | Description |
| -------- | ---- | ----------- |
| `textareaRef` | `RefObject<HTMLTextAreaElement>` | Ref to attach to the textarea |
| `adjustHeight` | `(reset?: boolean) => void` | Recalculate height; pass `true` to reset |
